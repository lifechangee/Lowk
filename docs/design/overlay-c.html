// Считаем EBITDA margin по сегментам
export async function buildReport(ctx: Ctx) {
  const rows = await ctx.db.query(`
    SELECT segment, revenue, opex, one_off
    FROM finance_q WHERE quarter = $1`, [ctx.q])
  return rows.map(r => ({
    segment: r.segment,
    margin: (r.revenue - r.opex + r.one_off) / r.revenue,
    cac: calcCac(r, ctx.partners),
    // TODO: вынести partner-канал в отдельную метрику
    flags: r.one_off > 0 ? ['one-off'] : [],
  }))
}
export const fmt = (x: number) => (x * 100).toFixed(1) + '%'
export function calcCac(r: Row, p: Partner[]) {
  return r.spend / Math.max(1, r.newCustomers)
}
